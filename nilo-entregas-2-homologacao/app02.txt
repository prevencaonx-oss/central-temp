function sla(d){if(['Programada','Reagendada','Cancelada','Retirada na loja'].includes(d.status))return {label:'Não aplicável',cls:'gray'};const end=d.finalized||isoNow();const t=mins(d.purchase,end);if(t>state.settings.slaClient)return{label:'Atrasado',cls:'red'};if(t>state.settings.slaClient*.85)return{label:'Atenção',cls:'orange'};return{label:'No prazo',cls:'green'}}
function calcKPIs(ds=todayDs()){
  const final=ds.filter(d=>d.status==='Finalizada');const orig=ds.filter(d=>!d.parentId);const success=orig.length?Math.round(final.filter(d=>d.attemptNo===1).length/orig.length*1000)/10:100;const ret=orig.length?Math.round(ds.filter(d=>d.status==='Devolvida').length/orig.length*1000)/10:0;
  const gross=ds.reduce((a,d)=>a+Number(d.tax||0),0),refund=ds.reduce((a,d)=>a+Number(d.refund||0),0),net=gross-refund,cost=state.costs.filter(c=>c.date===TODAY).reduce((a,c)=>a+Number(c.amount||0),0);
  const km=state.kmRecords.filter(k=>k.date===TODAY&&k.final!=null).reduce((a,k)=>a+(k.final-k.initial),0);
  const times=final.map(d=>mins(d.purchase,d.finalized));const avg=times.length?Math.round(times.reduce((a,b)=>a+b,0)/times.length):0;
  return{count:orig.length,final:final.length,success,ret,gross,refund,net,cost,balance:net-cost,km,avg,costPer:final.length?cost/final.length:0}
}
