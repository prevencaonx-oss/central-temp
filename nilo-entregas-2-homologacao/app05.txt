function bindRegisterForm(){const f=$('#registerForm');const renderTrips=()=>{const n=Number($('#tripCount').value||3);$('#tripTabs').innerHTML=Array.from({length:n},(_,i)=>`<div class="trip-card"><strong>Viagem ${i+1}</strong><small>ETA previsto</small><input class="form-control" style="margin-top:5px" name="tripEta${i+1}" type="time"><small>Observação</small><input class="form-control" style="margin-top:5px" name="tripObs${i+1}" placeholder="Itens desta viagem"></div>`).join('')};renderTrips();$$('input[name="large"]').forEach(r=>r.onchange=()=>{const large=f.large.value==='true';$('#largeBox').classList.toggle('hidden',!large);$('#optLarge').classList.toggle('active',large);$('#optNormal').classList.toggle('active',!large)});$('#tripCount').onchange=renderTrips;$$('input[name="scheduleMode"]').forEach(r=>r.onchange=()=>{const sch=f.scheduleMode.value==='scheduled';$('#scheduleBox').classList.toggle('hidden',!sch);$('#optScheduled').classList.toggle('active',sch);$('#optImmediate').classList.toggle('active',!sch)});f.onsubmit=e=>{e.preventDefault();const x=Object.fromEntries(new FormData(f));const pno=Number(x.purchaseNo),dupP=activeDeliveries().find(d=>d.purchaseNo===pno&&!d.parentId),dupC=activeDeliveries().find(d=>d.cupom===x.cupom&&!d.parentId);if(dupP)return toast('Número da compra já existe.');if(dupC&&!confirm('Este cupom já existe. Deseja registrar mesmo assim?'))return;state.seq++;const id=`DEL-${String(state.seq).padStart(6,'0')}`,purchase=`${x.date}T${x.time}`,large=x.large==='true',tripCount=large?Number(x.tripCount):1,trips=Array.from({length:tripCount},(_,i)=>({no:i+1,status:'Pendente',eta:x[`tripEta${i+1}`]||'',obs:x[`tripObs${i+1}`]||''})),scheduled=x.scheduleMode==='scheduled'?`${x.scheduledDate}T${x.scheduledTime}`:null;const d={id,purchaseNo:pno,cupom:x.cupom,doc:x.doc,pdv:x.pdv,purchase,client:x.client.trim().toUpperCase(),phone:formatPhone(x.phone),bairro:x.bairro,street:x.street.trim().toUpperCase(),number:x.number,complement:x.complement||'',reference:x.reference||'',priority:x.priority==='true',tax:Number(x.tax||0),type:scheduled?'Agendada':large?'Grande':'Normal',large,tripCount,trips,status:scheduled?'Programada':'Na loja',scheduled,rootId:id,parentId:null,attemptNo:1,driverId:'',vehicleId:'',cycleId:'',exit:null,arrival:null,finalized:null,returnTime:null,refund:0,reason:'',nextAction:'',notes:x.notes||'',history:[[fmtTime(purchase),scheduled?`Compra registrada e programada para ${fmtDateTime(scheduled)}`:'Compra registrada — aguardando saída']]};state.deliveries.push(d);audit('Cadastro de compra',id);closeModal();go('deliveries');toast('Compra registrada com sucesso')}}
function formatPhone(v){let n=String(v||'').replace(/\D/g,'');if(n.length===11)return`(${n.slice(0,2)}) ${n.slice(2,3)} ${n.slice(3,7)}-${n.slice(7)}`;if(n.length===10)return`(${n.slice(0,2)}) ${n.slice(2,6)}-${n.slice(6)}`;return String(v||'')}
function quickRegister(){const purchaseNo=Number($('#quickPurchase').value||0),cupom=$('#quickCupom').value.trim(),client=$('#quickClient').value.trim(),bairro=$('#quickBairro').value;if(!purchaseNo||!cupom||!client||!bairro)return toast('Preencha compra, cupom, cliente e bairro.');if(activeDeliveries().some(d=>d.purchaseNo===purchaseNo&&!d.parentId))return toast('Número de compra duplicado.');state.seq++;const id=`DEL-${String(state.seq).padStart(6,'0')}`,now=isoNow();state.deliveries.push({id,purchaseNo,cupom,doc:'',pdv:'',purchase:now,client:client.toUpperCase(),phone:'',bairro,street:'',number:'',complement:'',reference:'',priority:false,tax:0,type:'Normal',large:false,tripCount:1,trips:[{no:1,status:'Pendente',eta:'',obs:''}],status:'Na loja',scheduled:null,rootId:id,parentId:null,attemptNo:1,driverId:'',vehicleId:'',cycleId:'',exit:null,arrival:null,finalized:null,returnTime:null,refund:0,reason:'',nextAction:'',notes:'',history:[[fmtTime(now),'Compra registrada — lançamento rápido']]});audit('Cadastro rápido',id);render();toast('Compra registrada')}

function openDelivery(id){const d=activeDeliveries().find(x=>x.id===id);if(!d)return;const chain=activeDeliveries().filter(x=>x.rootId===d.rootId).sort((a,b)=>a.attemptNo-b.attemptNo);openModal(`<h2 class="modal-title">Compra Nº ${d.purchaseNo}</h2><p class="modal-sub">${d.id} • Tentativa ${d.attemptNo} • ${statusBadge(d.status)}</p><div class="grid-2"><div class="form-section"><h3>Dados da compra</h3><div class="metric-list"><div><small>Cupom</small><strong>${esc(d.cupom)}</strong></div><div><small>DOC</small><strong>${esc(d.doc||'—')}</strong></div><div><small>Caixa</small><strong>${esc(d.pdv||'—')}</strong></div><div><small>Taxa</small><strong>${money(d.tax)}</strong></div></div><p><b>${esc(d.client)}</b><br>${esc(d.phone)}<br>${esc(d.street)}, ${esc(d.number)} • ${esc(d.bairro)}<br>${esc(d.reference)}</p><div class="soft-note"><b>SLA:</b> ${statusBadge(sla(d).label)}<br>Compra → Saída: ${d.exit?duration(mins(d.purchase,d.exit)):'—'}<br>Loja → Cliente: ${d.exit&&(d.arrival||d.finalized)?duration(mins(d.exit,d.arrival||d.finalized)):'—'}<br>Compra → Cliente: ${(d.arrival||d.finalized)?duration(mins(d.purchase,d.arrival||d.finalized)):'—'}<br>Saída → Retorno: ${d.exit&&d.returnTime?duration(mins(d.exit,d.returnTime)):'—'}</div>${d.large?largeTripsHtml(d):''}</div><div class="form-section"><h3>Linha do tempo</h3><div class="trace-chain">${d.history.map(h=>`<div class="trace-item"><strong>${h[0]}</strong><small>${esc(h[1])}</small></div>`).join('')}</div><h3 style="margin-top:16px">Cadeia de tentativas</h3><div class="trace-chain">${chain.map(x=>`<div class="trace-item"><strong>Tentativa ${x.attemptNo} — ${x.id}</strong><small>${x.status} • ${fmtDateTime(x.purchase)}</small></div>`).join('')}</div></div></div><div class="modal-actions">${deliveryActions(d)}</div>`,true)}
function largeTripsHtml(d){return `<div class="trip-box" style="margin-top:12px"><b>Entrega grande — ${d.tripCount} viagens</b><div class="trip-tabs">${d.trips.map((t,i)=>`<div class="trip-card"><strong>Viagem ${t.no}</strong>${statusBadge(t.status)}<small>${t.eta?'ETA '+t.eta:''} ${esc(t.obs)}</small>${t.status!=='Concluída'?`<button class="green-btn sm-btn" style="margin-top:6px" onclick="completeTrip('${d.id}',${i})">Concluir viagem</button>`:''}</div>`).join('')}</div></div>`}
function deliveryActions(d){let a=`<button class="soft-btn" onclick="closeModal()">Fechar</button>`;if(d.status==='Na loja')a+=`<button class="blue-btn" onclick="closeModal();openCycleBuilder('${d.id}')">Adicionar à saída</button>`;if(d.status==='Em rota')a+=`<button class="soft-btn" onclick="markArrival('${d.id}')">Chegada ao cliente</button><button class="green-btn" onclick="finishDelivery('${d.id}')">Finalizar entrega</button><button class="danger-btn" onclick="returnDelivery('${d.id}')">Devolvida</button>`;if(['Programada','Reagendada'].includes(d.status))a+=`<button class="blue-btn" onclick="startScheduled('${d.id}')">Iniciar atendimento</button>`;if(d.status==='Devolvida')a+=`<button class="blue-btn" onclick="nextActionDialog('${d.id}')">Definir próxima ação</button>`;if(!['Finalizada','Cancelada'].includes(d.status))a+=`<button class="danger-btn" onclick="trashItem('delivery','${d.id}')">Mover para lixeira</button>`;return a}
function completeTrip(id,idx){const d=activeDeliveries().find(x=>x.id===id);if(!d)return;d.trips[idx].status='Concluída';d.history.push([new Date().toLocaleTimeString('pt-BR',{hour:'2-digit',minute:'2-digit'}),`Viagem ${idx+1} concluída`]);audit('Viagem concluída',`${id} / ${idx+1}`);openDelivery(id);toast('Viagem concluída')}
function markArrival(id){const d=activeDeliveries().find(x=>x.id===id);d.arrival=isoNow();d.history.push([fmtTime(d.arrival),'Chegada ao cliente registrada']);audit('Chegada ao cliente',id);openDelivery(id);toast('Chegada registrada')}
function finishDelivery(id){const d=activeDeliveries().find(x=>x.id===id);if(d.large&&d.trips.some(t=>t.status!=='Concluída'))return toast('Conclua todas as viagens da entrega grande antes de finalizar.');if(!d.arrival)d.arrival=isoNow();d.finalized=isoNow();d.status='Finalizada';if(d.trips.length)d.trips.forEach(t=>t.status='Concluída');d.history.push([fmtTime(d.finalized),'Entrega finalizada']);audit('Entrega finalizada',id);closeModal();render();toast('Entrega finalizada')}
function returnDelivery(id){const d=activeDeliveries().find(x=>x.id===id);openModal(`<h2 class="modal-title">Registrar devolução</h2><p class="modal-sub">${d.id} • ${d.client}</p><form id="returnForm"><div class="form-grid two"><label>Motivo<select class="form-control" name="reason">${state.reasons.map(x=>`<option>${x}</option>`).join('')}</select></label><label>Reembolso da taxa (R$)<input class="form-control" type="number" step="0.01" name="refund" value="${d.tax}"></label><label class="span-2">Observação<textarea class="form-control" name="obs"></textarea></label></div><div class="modal-actions"><button type="button" class="soft-btn" onclick="openDelivery('${id}')">Voltar</button><button class="danger-btn">Registrar devolvida</button></div></form>`);$('#returnForm').onsubmit=e=>{e.preventDefault();const x=Object.fromEntries(new FormData(e.target));d.status='Devolvida';d.reason=x.reason;d.refund=Number(x.refund||0);d.returnTime=isoNow();d.nextAction='';d.history.push([fmtTime(d.returnTime),`Devolvida: ${x.reason}${x.obs?' — '+x.obs:''}`]);audit('Entrega devolvida',id);closeModal();render();toast('Devolução registrada')}}
