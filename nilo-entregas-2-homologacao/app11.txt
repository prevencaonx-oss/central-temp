function hubSmartOrder(cycle){
  if(!cycle)return[];
  return cycle.deliveryIds.map(id=>activeDeliveries().find(d=>d.id===id)).filter(Boolean).sort((a,b)=>{
    const pr=(b.priority?1:0)-(a.priority?1:0);if(pr)return pr;
    const ba=bairroOrder(a.bairro)-bairroOrder(b.bairro);if(ba)return ba;
    return String(a.purchase||'').localeCompare(String(b.purchase||''));
  });
}
function hubRouteReason(d){if(d.priority)return 'Prioridade';const order=bairroOrder(d.bairro);return `Bairro • ordem ${order}`}

function hubInRoutePanel(inRoute){
  const activeCycles=state.cycles.filter(c=>c.status==='Em andamento').sort((a,b)=>String(a.exit).localeCompare(String(b.exit)));
  const ids=new Set(),ordered=[];activeCycles.forEach(c=>hubSmartOrder(c).forEach(d=>{if(d.status==='Em rota'&&!ids.has(d.id)){ids.add(d.id);ordered.push(d)}}));inRoute.forEach(d=>{if(!ids.has(d.id)){ids.add(d.id);ordered.push(d)}});
  return `<section class="panel hub-route-queue-panel"><div class="panel-title"><div><h3>Entregas em rota</h3><p>Todas as entregas que estão na rua. Clique em qualquer uma para registrar a chegada.</p></div><span class="badge orange">${ordered.length} em rota</span></div><div class="hub-route-queue-list">${ordered.length?ordered.map(hubRouteQueueRow).join(''):`<div class="hub-route-empty">${hubIcon('truck')}<b>Nenhuma entrega em rota agora</b><small>Quando uma saída for iniciada, todas as entregas aparecerão aqui.</small></div>`}</div></section>`
}
function googleHubMapCentral(cycle){
  if(!cycle)return `<div class="google-map-empty"><div class="stat-icon blue">${hubIcon('map')}</div><b>Nenhuma rota em andamento agora</b><small>Assim que uma nova saída for iniciada, o Google Maps aparecerá aqui.</small><button class="blue-btn" onclick="hubNewRoute()">Nova saída</button></div>`;
  const ds=hubSmartOrder(cycle),current=ds.find(d=>d.status==='Em rota')||ds.find(d=>d.status!=='Finalizada'&&d.status!=='Devolvida')||ds[ds.length-1];
