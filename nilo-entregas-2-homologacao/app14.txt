function hubStartOperation(){
  if(!state.closedDay)return toast('A operação do dia já está em andamento.');
  state.closedDay=false;
  const last=state.closings.at(-1);if(last&&!last.reopenedAt)last.reopenedAt=isoNow();
  audit('Operação do dia iniciada / reaberta',TODAY);render();toast('Operação iniciada')
}
function hubEndOperation(){
  if(state.closedDay)return toast('A operação do dia já está encerrada.');
  closeDayDialog();hubStyleModal('green')
}
function hubCentralSearch(){
  const input=document.getElementById('hubCentralSearch');
  const q=String(input?.value||'').trim().toUpperCase();
  if(!q){input?.focus();return toast('Digite uma informação para pesquisar.');}
  const ds=activeDeliveries().filter(d=>[d.id,d.purchaseNo,d.cupom,d.doc,d.pdv,d.client,d.phone,d.bairro,d.street,d.number,d.cycleId,d.driverId?driverName(d.driverId):'',d.status].some(v=>String(v||'').toUpperCase().includes(q)));
  openModal(`<div class="hub-search-modal-head">${hubIcon('package')}<div><small>PESQUISA DA CENTRAL</small><h2 class="modal-title">Resultados para “${esc(q)}”</h2><p class="modal-sub">${ds.length} entrega(s) encontrada(s).</p></div></div>${ds.length?deliveryTable(ds.slice(0,30)):`<div class="empty">Nenhuma entrega encontrada.</div>`}<div class="modal-actions"><button class="soft-btn" onclick="closeModal()">Fechar</button><button class="blue-btn" onclick="closeModal();go('search')">Pesquisa avançada</button></div>`,true);hubStyleModal('blue')
}
function hubOperationConsole(){
  return `<section class="hub-operation-console">
    <button class="hub-op-btn hub-op-start ${!state.closedDay?'is-current':''}" onclick="hubStartOperation()" ${!state.closedDay?'aria-disabled="true"':''}>
      <span class="hub-op-dot"></span><span><b>Iniciar operação</b><small>${state.closedDay?'Liberado':'Em andamento'}</small></span>
    </button>
    <button class="hub-op-btn hub-op-end ${state.closedDay?'is-disabled':''}" onclick="hubEndOperation()" ${state.closedDay?'aria-disabled="true"':''}>
      ${hubIcon('stop')}<span><b>Encerrar operação</b><small>${state.closedDay?'Encerrada':'Fechamento do dia'}</small></span>
    </button>
    <div class="hub-central-searchbar">${hubIcon('package')}<input id="hubCentralSearch" placeholder="Pesquisar entrega: compra, cupom, DOC, caixa, cliente, telefone, bairro, entregador ou ciclo..." onkeydown="if(event.key==='Enter')hubCentralSearch()"><button onclick="hubCentralSearch()">Pesquisar</button></div>
  </section>`
}

function hubKmVehicleHero(v,mode='open',record=null){
  return `<div class="hub-km-hero"><div class="hub-km-icon">${hubIcon('gauge')}</div><div class="hub-km-vehicle"><small>VEÍCULO</small><h3>${esc(v.name)} <span>• ${esc(v.plate)}</span></h3><p>${esc(v.type||'Veículo da frota')}</p></div><div class="hub-km-current"><small>${mode==='open'?'Odômetro cadastrado':'KM inicial'}</small><strong>${num(mode==='open'?v.odometer:record.initial)} km</strong></div></div>`
}
function openKm(vehicleId){
  const v=state.vehicles.find(x=>x.id===vehicleId);if(!v)return;
  openModal(`<div class="hub-km-title"><span>QUILOMETRAGEM & FROTA</span><h2 class="modal-title">Abrir expediente de quilometragem</h2><p class="modal-sub">Registre o odômetro no início do expediente para calcular KM diário, por ciclo e por entrega.</p></div>${hubKmVehicleHero(v,'open')}<form id="kmOpen"><div class="form-section hub-km-form"><h3>Leitura inicial</h3><div class="form-grid two"><label>KM inicial do expediente<input class="form-control hub-km-big-input" name="initial" type="number" min="0" value="${v.odometer}" required></label><div class="hub-km-tip"><b>Conferência rápida</b><small>Digite exatamente o valor mostrado no painel do veículo antes da primeira saída.</small></div></div></div><div class="modal-actions"><button type="button" class="soft-btn" onclick="closeModal()">Cancelar</button><button class="blue-btn">Abrir expediente</button></div></form>`,false);hubStyleModal('teal');
  document.getElementById('kmOpen').onsubmit=e=>{e.preventDefault();const initial=Number(new FormData(e.target).get('initial'));if(initial<0)return toast('Informe um KM válido.');state.kmRecords.push({id:uid('KM'),date:TODAY,vehicleId,initial,final:null});audit('KM inicial registrado',vehicleId);closeModal();render();toast('Expediente de KM aberto')}
}
function closeKm(id){
  const r=state.kmRecords.find(x=>x.id===id),v=state.vehicles.find(x=>x.id===r?.vehicleId);if(!r||!v)return;
  openModal(`<div class="hub-km-title"><span>QUILOMETRAGEM & FROTA</span><h2 class="modal-title">Fechar quilometragem do dia</h2><p class="modal-sub">Informe o odômetro final para calcular automaticamente o total rodado.</p></div>${hubKmVehicleHero(v,'close',r)}<form id="kmClose"><div class="form-section hub-km-form"><h3>Leitura final</h3><div class="form-grid two"><label>KM inicial<input class="form-control" value="${r.initial}" disabled></label><label>KM final<input class="form-control hub-km-big-input" name="final" type="number" min="${r.initial}" required></label></div><div id="hubKmPreview" class="hub-km-preview"><small>Total rodado</small><strong>— km</strong></div></div><div class="modal-actions"><button type="button" class="soft-btn" onclick="closeModal()">Cancelar</button><button class="blue-btn">Fechar quilometragem</button></div></form>`,false);hubStyleModal('teal');
  const f=document.querySelector('#kmClose [name="final"]');f.addEventListener('input',()=>{const x=Number(f.value||0),p=document.getElementById('hubKmPreview');p.innerHTML=`<small>Total rodado</small><strong>${x>=r.initial?num(x-r.initial):'—'} km</strong>`});
  document.getElementById('kmClose').onsubmit=e=>{e.preventDefault();const final=Number(new FormData(e.target).get('final'));if(final<r.initial)return toast('KM final não pode ser menor que o inicial.');r.final=final;v.odometer=final;audit('KM diário fechado',v.id);closeModal();render();toast('Quilometragem fechada')}
}

function hubCycleModalMap(c){
  const ds=hubSmartOrder(c);const current=ds.find(d=>d.status==='Em rota')||ds[0];
  const addr=current?`${current.street||''}, ${current.number||''}, ${current.bairro||''}, Nova Xavantina - MT`:state.settings.origin;
  const src=`https://maps.google.com/maps?hl=pt-BR&z=14&output=embed&q=${encodeURIComponent(addr)}`;
  return `<div class="hub-cycle-map"><iframe src="${src}" loading="lazy" referrerpolicy="no-referrer-when-downgrade" title="Google Maps do ciclo ${c.id}"></iframe><span class="hub-cycle-map-label">Google Maps</span></div><div class="route-sequence hub-cycle-sequence">${ds.map((d,i)=>`<span class="route-chip ${d.status==='Finalizada'?'done':d.status==='Em rota'?'active':''}"><b>${i+1}</b>${esc(d.bairro)} <small>${esc(d.client)}</small></span>`).join('')}</div>`
}
