function hubKmDraftCapture(){
  const d=window.__hubKmFleetDraft;if(!d)return;
  d.route.forEach(id=>{const el=document.getElementById(`hubKmInitial_${id}`);if(el)d.initials[id]=Number(el.value||0)});
}
function hubKmVehicleChoice(v,kind,selected){
  const fn=kind==='route'?'Rota':'Fechamento';
  return `<button type="button" class="hub-km-choice ${selected?'selected':''}" onclick="hubKmToggle('${kind}','${v.id}')">
    <span class="hub-km-choice-check">${selected?'✓':''}</span>
    <span class="hub-km-choice-icon">${hubIcon(v.type==='Moto'?'truck':'gauge')}</span>
    <span><b>${esc(v.name)}</b><small>${esc(v.plate)} • ${esc(v.type||'Veículo')}</small></span>
    <em>${fn}</em>
  </button>`
}
function hubKmToggle(kind,id){
  hubKmDraftCapture();
  const d=window.__hubKmFleetDraft;if(!d)return;
  const arr=d[kind];const i=arr.indexOf(id);if(i>=0)arr.splice(i,1);else arr.push(id);
  if(kind==='route'&&arr.includes(id)&&!d.initials[id]){const v=state.vehicles.find(x=>x.id===id);d.initials[id]=Number(v?.odometer||0)}
  hubRenderKmFleetModal();
}

function hubKmSaveFleetOpen(){
  hubKmDraftCapture();const d=window.__hubKmFleetDraft;if(!d)return;
  if(!d.route.length)return toast('Selecione pelo menos um veículo para a rota.');
  if(!d.closing.length)return toast('Selecione pelo menos um veículo para o fechamento do dia.');
  state.settings.routeVehicleIds=[...d.route];state.settings.closingVehicleIds=[...d.closing];
  d.route.forEach(vehicleId=>{
    const initial=Number(d.initials[vehicleId]||0);if(initial<0)return;
    const existing=state.kmRecords.find(r=>r.date===TODAY&&r.vehicleId===vehicleId&&r.final==null);
    if(existing)existing.initial=initial;else state.kmRecords.push({id:uid('KM'),date:TODAY,vehicleId,initial,final:null});
  });
  audit('Veículos do expediente definidos',`Rota: ${d.route.join(', ')} | Fechamento: ${d.closing.join(', ')}`);
  window.__hubKmFleetDraft=null;closeModal();render();toast('Veículos do expediente configurados');
}
function openKm(vehicleId){
  const active=state.vehicles.filter(v=>v.active);if(!active.length)return toast('Nenhum veículo ativo cadastrado.');
  const openToday=state.kmRecords.filter(r=>r.date===TODAY&&r.final==null).map(r=>r.vehicleId);
  const routeSaved=Array.isArray(state.settings.routeVehicleIds)?state.settings.routeVehicleIds:[];
  const closeSaved=Array.isArray(state.settings.closingVehicleIds)?state.settings.closingVehicleIds:[];
  const first=vehicleId||routeSaved[0]||openToday[0]||active[0].id;
  const route=[...new Set(routeSaved.length?routeSaved:(openToday.length?openToday:[first]))].filter(id=>active.some(v=>v.id===id));
  const closing=[...new Set(closeSaved.length?closeSaved:route)].filter(id=>active.some(v=>v.id===id));
  const initials={};route.forEach(id=>{const r=state.kmRecords.find(x=>x.date===TODAY&&x.vehicleId===id&&x.final==null),v=state.vehicles.find(x=>x.id===id);initials[id]=Number(r?.initial??v?.odometer??0)});
  window.__hubKmFleetDraft={route,closing,initials};
  openModal('<div></div>',true);hubRenderKmFleetModal();
}

function central(){
  const ds=todayDs(),k=calcKPIs(ds);
  const waiting=ds.filter(d=>d.status==='Na loja');
  const inRoute=ds.filter(d=>d.status==='Em rota');
  const critical=ds.filter(d=>d.status==='Devolvida'||sla(d).cls==='red');
  const scheduled=ds.filter(d=>d.status==='Programada');
  const closedKm=state.kmRecords.filter(x=>x.date===TODAY&&x.final!=null);
  const openCycles=state.cycles.filter(c=>c.status==='Em andamento');

  const usedVehicleIds=[...new Set(todayCycles.map(c=>c.vehicleId).filter(Boolean))];
  const usedVehicles=usedVehicleIds.map(id=>state.vehicles.find(v=>v.id===id)).filter(Boolean);
  const delivered=ds.filter(d=>d.status==='Finalizada').length;
  const deliveredPct=ds.length?Math.round(delivered/ds.length*100):0;

  const grossGains=ds.reduce((a,d)=>a+Number(d.tax||0),0);
  const refunds=ds.reduce((a,d)=>a+Number(d.refund||0),0);
  const dayGains=grossGains-refunds;
  const delayed=ds.filter(d=>!d.scheduled&&mins(d.purchase,d.arrival||d.finalized||isoNow())>Number(state.settings.slaClient||210));
  const dayResult=dayGains-dayCosts;
  const waitPop=waiting.slice(0,6).map(d=>hubListItem('package',`Compra Nº ${d.purchaseNo} — ${d.client}`,`${d.bairro} • aguardando ${duration(mins(d.purchase))}`)).join('');

  const attentionPop=critical.slice(0,6).map(d=>hubListItem('alert',`Compra Nº ${d.purchaseNo} — ${d.client}`,`${d.bairro} • ${d.status==='Devolvida'?(d.reason||'Devolvida'):sla(d).label}`)).join('');
  const schedPop=scheduled.slice(0,6).map(d=>hubListItem('clock',`Compra Nº ${d.purchaseNo} — ${d.client}`,`${fmtDateTime(d.scheduled)} • ${d.bairro}`)).join('');
  const kmPop=closedKm.map(r=>{const v=state.vehicles.find(x=>x.id===r.vehicleId);return hubListItem('gauge',v?`${v.name} — ${v.plate}`:r.vehicleId,`${num(r.final-r.initial)} km rodados`)}).join('');
  const vehiclePop=usedVehicles.map(v=>{const cs=todayCycles.filter(c=>c.vehicleId===v.id);const active=cs.some(c=>c.status==='Em andamento');return hubListItem('truck',`${v.name} — ${v.plate}`,`${cs.length} ciclo(s) hoje • ${active?'em operação':'sem ciclo ativo'}`)}).join('');
  const deliveredPop=ds.filter(d=>d.status==='Finalizada').slice(-8).reverse().map(d=>hubListItem('package',`Compra Nº ${d.purchaseNo} — ${d.client}`,`${d.bairro} • entregue ${fmtTime(d.finalized||d.arrival)}`,'green')).join('');
  const costPop=state.costs.filter(c=>c.date===TODAY).map(c=>hubListItem('gauge',c.category,`${vehicleName(c.vehicleId)} • ${money(c.amount)}`)).join('');
  const gainPop=`${hubListItem('package','Taxas cobradas no dia',money(grossGains))}${hubListItem('alert','Reembolsos/devoluções',money(refunds))}${hubListItem('package','Ganhos líquidos',money(dayGains),'green')}`;
  const delayedPop=delayed.slice(0,8).map(d=>hubListItem('alert',`Compra Nº ${d.purchaseNo} — ${d.client}`,`${d.bairro} • ${duration(mins(d.purchase,d.arrival||d.finalized||isoNow()))}`,'red')).join('');
  const resultPop=`${hubListItem('package','Ganhos do dia',money(dayGains),'green')}${hubListItem('gauge','Custos do dia',money(dayCosts),'red')}${hubListItem(dayResult>=0?'package':'alert',dayResult>=0?'Lucro operacional':'Prejuízo operacional',money(Math.abs(dayResult)),dayResult>=0?'green':'red')}`;
  return `<div class="page-grid hub-central hub-central-v8 hub-central-v9 hub-central-v10">
    ${hubOperationConsole()}
    <div class="stats hub-stats hub-stats-v9">
      ${hubLiveStat('package','blue','Aguardando coleta',waiting.length,'Compra(s) pronta(s) na loja','Entregas aguardando coleta',waitPop,'deliveries')}
      ${hubLiveStat('truck','orange','Em rota',inRoute.length,`${openCycles.length} saída(s) em andamento`,'Entregas que estão na rua',routePop,'routes')}

      ${hubLiveStat('clock','red','Entregas atrasadas',delayed.length,'Fora do SLA do dia','Entregas atrasadas',delayedPop||'<div class="empty">Nenhuma entrega atrasada.</div>','deliveries')}
