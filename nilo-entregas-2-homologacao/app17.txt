function hubKmOpenVehicleIds(){return state.kmRecords.filter(r=>r.date===TODAY&&r.final==null).map(r=>r.vehicleId)}
function hubKmRetireVehicle(vehicleId){
  const r=state.kmRecords.find(x=>x.date===TODAY&&x.vehicleId===vehicleId&&x.final==null),v=state.vehicles.find(x=>x.id===vehicleId);if(!r||!v)return toast('Veículo sem expediente aberto.');
  openModal(`<div class="hub-km-title"><span>QUILOMETRAGEM & FROTA</span><h2 class="modal-title">Retirar veículo da operação</h2><p class="modal-sub">Informe o KM final somente agora, no momento em que o veículo deixar a operação.</p></div><div class="hub-retire-vehicle"><div class="hub-retire-icon">${hubIcon('truck')}</div><div><small>VEÍCULO</small><h3>${esc(v.name)} <span>• ${esc(v.plate)}</span></h3><p>KM inicial do expediente: <b>${num(r.initial)} km</b></p></div></div><form id="hubKmRetireForm"><div class="form-section hub-km-form"><h3>KM final ao retirar da operação</h3><div class="form-grid two"><label>KM inicial<input class="form-control" value="${r.initial}" disabled></label><label>KM final<input class="form-control hub-km-big-input" name="final" type="number" min="${r.initial}" required autofocus></label></div><div id="hubKmRetirePreview" class="hub-km-preview"><small>Total rodado</small><strong>— km</strong></div></div><div class="modal-actions"><button type="button" class="soft-btn" onclick="closeModal()">Cancelar</button><button class="green-btn">Retirar veículo e fechar KM</button></div></form>`,false);hubStyleModal('teal');
  const f=document.querySelector('#hubKmRetireForm [name="final"]');f.addEventListener('input',()=>{const x=Number(f.value||0);document.getElementById('hubKmRetirePreview').innerHTML=`<small>Total rodado</small><strong>${x>=r.initial?num(x-r.initial):'—'} km</strong>`});
  document.getElementById('hubKmRetireForm').onsubmit=e=>{e.preventDefault();const final=Number(new FormData(e.target).get('final'));if(final<r.initial)return toast('KM final não pode ser menor que o inicial.');r.final=final;v.odometer=final;state.settings.routeVehicleIds=(state.settings.routeVehicleIds||[]).filter(id=>id!==vehicleId);state.settings.closingVehicleIds=(state.settings.closingVehicleIds||[]).filter(id=>id!==vehicleId);audit('Veículo retirado da operação / KM fechado',vehicleId);closeModal();render();toast(`${v.name} retirado da operação`)}
}
